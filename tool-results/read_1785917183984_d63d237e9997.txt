   153→function getDeadlineCountdown(deadline: Date | string | null): string {
   154→  if (!deadline) return ''
   155→  const d = typeof deadline === 'string' ? parseISO(deadline) : deadline
   156→  const now = new Date()
   157→  const days = differenceInDays(d, now)
   158→
   159→  if (days < 0) return `${Math.abs(days)} days overdue`
   160→  if (days === 0) return 'Due today'
   161→  if (days === 1) return '1 day left'
   162→  if (days <= 7) return `${days} days left`
   163→  if (days <= 30) {
   164→    const weeks = Math.floor(days / 7)
   165→    const remDays = days % 7
   166→    return remDays > 0
   167→      ? `${weeks}w ${remDays}d left`
   168→      : `${weeks} week${weeks > 1 ? 's' : ''} left`
   169→  }
   170→  const months = differenceInMonths(d, now)
   171→  if (months > 0) {
   172→    const remDays = differenceInDays(d, addMonths(now, months))
   173→    return remDays > 0
   174→      ? `${months}mo ${remDays}d left`
   175→      : `${months} month${months > 1 ? 's' : ''} left`
   176→  }
   177→  return `${days} days left`
   178→}

   180→function estimateCompletionDate(
   181→  currentAmount: number,
   182→  targetAmount: number,
   183→  createdAt: Date | string,
   184→): string | null {
   185→  if (currentAmount <= 0 || targetAmount <= 0) return null
   186→  if (currentAmount >= targetAmount) return null
   187→
   188→  const created = typeof createdAt === 'string' ? parseISO(createdAt) : createdAt
   189→  const now = new Date()
   190→  const daysElapsed = Math.max(differenceInDays(now, created), 1)
   191→  const dailyRate = currentAmount / daysElapsed
   192→
   193→  if (dailyRate <= 0) return null
   194→
   195→  const remaining = targetAmount - currentAmount
   196→  const daysNeeded = Math.ceil(remaining / dailyRate)
   197→  return format(addDays(now, daysNeeded), 'MMM d, yyyy')
   198→}

   210→// ---------------------------------------------------------------------------
   211→// Circular Progress
   212→// ---------------------------------------------------------------------------
   213→
   214→function CircularProgress({
   215→  value,
   216→  size = 80,
   217→  strokeWidth = 6,
   218→  color = '#10b981',
   219→}: {
   220→  value: number
   221→  size?: number
   222→  strokeWidth?: number
   223→  color?: string
   224→}) {
   225→  const radius = (size - strokeWidth) / 2
   226→  const circumference = 2 * Math.PI * radius
   227→  const clampedValue = Math.min(Math.max(value, 0), 100)
   228→  const offset = circumference - (clampedValue / 100) * circumference
   229→
   230→  return (
   231→    <div className="relative" style={{ width: size, height: size }}>
   232→      <svg
   233→        width={size}
   234→        height={size}
   235→        className="-rotate-90"
   236→        aria-label={`${Math.round(clampedValue)}% complete`}
   237→      >
   238→        <circle
   239→          cx={size / 2}
   240→          cy={size / 2}
   241→          r={radius}
   242→          fill="none"
   243→          stroke="currentColor"
   244→          strokeWidth={strokeWidth}
   245→          className="text-muted/30"
   246→        />
   247→        <motion.circle
   248→          cx={size / 2}
   249→          cy={size / 2}
   250→          r={radius}
   251→          fill="none"
   252→          stroke={color}
   253→          strokeWidth={strokeWidth}
   254→          strokeLinecap="round"
   255→          strokeDasharray={circumference}
   256→          initial={{ strokeDashoffset: circumference }}
   257→          animate={{ strokeDashoffset: offset }}
   258→          transition={{ duration: 1, ease: 'easeOut', delay: 0.3 }}
   259→        />
   260→      </svg>
   261→      <div className="absolute inset-0 flex items-center justify-center">
   262→        <span className="text-sm font-bold">
   263→          {Math.round(clampedValue)}%
   264→        </span>
   265→      </div>
   266→    </div>
   267→  )
   268→}
