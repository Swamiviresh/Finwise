   157→
   158→export function TransactionsPage() {
   159→  const { user } = useRouterStore()
   160→  const isMobile = useMediaQuery('(max-width: 768px)')
   161→  const currency = user?.currency || APP_CONFIG.DEFAULT_CURRENCY
   162→
   163→  // ---- Filters state ----
   164→  const [filters, setFilters] = React.useState<TransactionFilters>({
   165→    page: 1,
   166→    limit: 10,
   167→    sortBy: 'date',
   168→    sortOrder: 'desc',
   169→  })
   170→  const [searchInput, setSearchInput] = React.useState('')
   171→  const debounceRef = React.useRef<NodeJS.Timeout | null>(null)
   172→
   173→  const updateFilter = React.useCallback(
   174→    (patch: Partial<TransactionFilters>) => {
   175→      setFilters((prev) => ({
   176→        ...prev,
   177→        ...patch,
   178→        // Reset to page 1 when any filter changes (except page itself)
   179→        page: patch.page !== undefined ? patch.page : 1,
   180→      }))
   181→    },
   182→    [],
   183→  )
   184→
   185→  const clearFilters = React.useCallback(() => {
   186→    setFilters({
   187→      page: 1,
   188→      limit: filters.limit,
   189→      sortBy: 'date',
   190→      sortOrder: 'desc',
   191→    })
   192→    setSearchInput('')
   193→    setStartDate(undefined)
   194→    setEndDate(undefined)
   195→  }, [filters.limit])
   196→
   197→  const hasActiveFilters = React.useMemo(
   198→    () =>
   199→      !!filters.search ||
   200→      !!filters.type ||
   201→      !!filters.categoryId ||
   202→      !!filters.startDate ||
   203→      !!filters.endDate ||
   204→      filters.sortBy !== 'date' ||
   205→      filters.sortOrder !== 'desc',
   206→    [filters],
   207→  )
   208→
   209→  // ---- Debounced search ----
   210→  const handleSearchChange = React.useCallback(
   211→    (value: string) => {
   212→      setSearchInput(value)
   213→      if (debounceRef.current) clearTimeout(debounceRef.current)
   214→      debounceRef.current = setTimeout(() => {
   215→        updateFilter({ search: value || undefined })
   216→      }, 300)
   217→    },
   218→    [updateFilter],
   219→  )
   220→
   221→  // ---- Date state ----
   222→  const [startDate, setStartDate] = React.useState<Date | undefined>(undefined)
   223→  const [endDate, setEndDate] = React.useState<Date | undefined>(undefined)
   224→  const [startOpen, setStartOpen] = React.useState(false)
   225→  const [endOpen, setEndOpen] = React.useState(false)
   226→
   227→  React.useEffect(() => {
   228→    updateFilter({
   229→      startDate: startDate ? format(startDate, 'yyyy-MM-dd') : undefined,
   230→      endDate: endDate ? format(endDate, 'yyyy-MM-dd') : undefined,
   231→    })
   232→    }, [startDate, endDate])
   233→
   234→  // ---- Dialog / Sheet state ----
   235→  const [formOpen, setFormOpen] = React.useState(false)
   236→  const [editingTransaction, setEditingTransaction] =
   237→    React.useState<TransactionWithCategory | null>(null)
   238→  const [deleteTarget, setDeleteTarget] =
   239→    React.useState<TransactionWithCategory | null>(null)
   240→
   241→  // ---- Data ----
   242→  const { data, isLoading, isError } = useTransactions(filters)
   243→
   244→  const categoriesQuery = useQuery<ApiResponse<Category>>({
   245→    queryKey: ['categories'],
   246→    queryFn: () => apiRequest('/api/categories'),
   247→    staleTime: 5 * 60 * 1000,
   248→  })
   249→  const categories = React.useMemo(() => {
   250→    if (categoriesQuery.data?.data) {
   251→      return categoriesQuery.data.data as unknown as Category[]
   252→    }
   253→    return DEFAULT_CATEGORIES.map((c) => ({
   254→      id: c.name.toLowerCase().replace(/\s+/g, '-'),
   255→      name: c.name,
   256→      icon: c.icon,
   257→      color: c.color,
   258→      type: c.type,
   259→      isDefault: true,
   260→      userId: null,
   261→      createdAt: new Date(),
   262→      updatedAt: new Date(),
   263→    })) as Category[]
   264→  }, [categoriesQuery.data])
   265→
   266→  const transactions = data?.data ?? []
   267→  const pagination = data?.pagination
   268→  const totalItems = pagination?.total ?? 0
   269→  const totalPages = pagination?.totalPages ?? 0
   270→  const currentPage = filters.page ?? 1
   271→  const currentLimit = filters.limit ?? 10
   272→
   273→  const startIdx = (currentPage - 1) * currentLimit + 1
   274→  const endIdx = Math.min(currentPage * currentLimit, totalItems)
   275→
   276→  // ---- Mutations ----
   277→  const createMutation = useCreateTransaction()
   278→  const updateMutation = useUpdateTransaction()
   279→  const deleteMutation = useDeleteTransaction()
   280→
   281→  const isSubmitting =
   282→    createMutation.isPending || updateMutation.isPending
   283→
   284→  // ---- Form ----
   285→  const form = useForm<TransactionFormValues>({
   286→    resolver: zodResolver(formSchema),
   287→    defaultValues: {
   288→      amount: undefined as unknown as number,
   289→      description: '',
   290→      type: 'expense',
   291→      date: format(new Date(), 'yyyy-MM-dd'),
   292→      categoryId: '',
   293→      paymentMethod: '',
   294→      notes: '',
   295→    },
   296→  })
   297→
   298→  const watchedType = form.watch('type')
   299→
   300→  const filteredCategories = React.useMemo(() => {
   301→    if (!watchedType) return categories
   302→    return categories.filter(
   303→      (c) => c.type === watchedType || c.type === 'transfer',
   304→    )
   305→  }, [categories, watchedType])
   306→
   307→  const openCreateForm = React.useCallback(() => {
   308→    setEditingTransaction(null)
   309→    form.reset({
   310→      amount: undefined as unknown as number,
   311→      description: '',
   312→      type: 'expense',
   313→      date: format(new Date(), 'yyyy-MM-dd'),
   314→      categoryId: '',
   315→      paymentMethod: '',
   316→      notes: '',
   317→    })
   318→    setFormOpen(true)
   319→  }, [form])
   320→
   321→  const openEditForm = React.useCallback(
   322→    (t: TransactionWithCategory) => {
   323→      setEditingTransaction(t)
   324→      form.reset({
   325→        amount: t.amount,
   326→        description: t.description,
   327→        type: t.type,
   328→        date: format(new Date(t.date), 'yyyy-MM-dd'),
   329→        categoryId: t.categoryId,
   330→        paymentMethod: t.paymentMethod || '',
   331→        notes: t.notes || '',
   332→      })
   333→      setFormOpen(true)
   334→    },
   335→    [form],
   336→  )
   337→
   338→  const onSubmit = React.useCallback(
   339→    (values: TransactionFormValues) => {
   340→      if (editingTransaction) {
   341→        updateMutation.mutate(
   342→          { id: editingTransaction.id, data: values },
   343→          { onSuccess: () => setFormOpen(false) },
   344→        )
   345→      } else {
   346→        createMutation.mutate(values, {
   347→          onSuccess: () => setFormOpen(false),
   348→        })
   349→      }
   350→    },
   351→    [editingTransaction, createMutation, updateMutation],
   352→  )
   353→
   354→  const handleDelete = React.useCallback(() => {
   355→    if (!deleteTarget) return
   356→    deleteMutation.mutate(deleteTarget.id, {
   357→      onSuccess: () => setDeleteTarget(null),
   358→    })
   359→  }, [deleteTarget, deleteMutation])
   360→
   361→  // ---- Export ----
   362→  const exportTransactions = React.useCallback(
   363→    (format: 'csv' | 'json') => {
   364→      if (transactions.length === 0) return
   365→
   366→      let content: string
   367→      let mimeType: string
   368→      let ext: string
   369→
   370→      if (format === 'json') {
   371→        const exportData = transactions.map((t) => ({
   372→          description: t.description,
   373→          amount: t.amount,
   374→          type: t.type,
   375→          category: t.category?.name || '',
   376→          paymentMethod: t.paymentMethod || '',
   377→          date: t.date,
   378→          notes: t.notes || '',
   379→        }))
   380→        content = JSON.stringify(exportData, null, 2)
   381→        mimeType = 'application/json'
   382→        ext = 'json'
   383→      } else {
   384→        const headers = [
   385→          'Description',
   386→          'Amount',
   387→          'Type',
   388→          'Category',
   389→          'Payment Method',
   390→          'Date',
   391→          'Notes',
   392→        ]
   393→        const rows = transactions.map((t) => [
   394→          `"${t.description.replace(/"/g, '\"')}"`,
   395→          t.amount.toFixed(2),
   396→          t.type,
   397→          t.category?.name || '',
   398→          t.paymentMethod || '',
   399→          format(new Date(t.date), 'yyyy-MM-dd'),
   400→          `"${(t.notes || '').replace(/"/g, '\"')}"`,
   401→        ])
   402→        content = [headers.join(','), ...rows.map((r) => r.join(','))].join(
   403→          '\n',
   404→        )
   405→        mimeType = 'text/csv'
   406→        ext = 'csv'
   407→      }
   408→
   409→      const blob = new Blob([content], { type: mimeType })
   410→      const url = URL.createObjectURL(blob)
   411→      const a = document.createElement('a')
   412→      a.href = url
   413→      a.download = `transactions.${ext}`
   414→      a.click()
   415→      URL.revokeObjectURL(url)
   416→    },
   417→    [transactions],
   418→  )
   419→
   420→  // ---- Sort toggle ----
   421→  const toggleSortOrder = React.useCallback(() => {
   422→    updateFilter({
   423→      sortOrder: filters.sortOrder === 'asc' ? 'desc' : 'asc',
   424→    })
   425→  }, [filters.sortOrder, updateFilter])
   426→
   427→  const SortIcon =
   428→    filters.sortOrder === 'asc' ? ArrowUp : ArrowDown
   429→
   430→  // ---- Page numbers ----
   431→  const pageNumbers = React.useMemo(() => {
   432→    const pages: (number | '...')[] = []
   433→    if (totalPages <= 7) {
   434→      for (let i = 1; i <= totalPages; i++) pages.push(i)
   435→    } else {
   436→      pages.push(1)
   437→      if (currentPage > 3) pages.push('...')
   438→      const start = Math.max(2, currentPage - 1)
   439→      const end = Math.min(totalPages - 1, currentPage + 1)
   440→      for (let i = start; i <= end; i++) pages.push(i)
   441→      if (currentPage < totalPages - 2) pages.push('...')
   442→      pages.push(totalPages)
   443→    }
   444→    return pages
   445→  }, [totalPages, currentPage])
   446→
   447→  // ---- Get payment method label ----
   448→  const getPaymentLabel = React.useCallback(
   449→    (val: string | null) => {
   450→      if (!val) return '—'
   451→      return PAYMENT_METHODS.find((m) => m.value === val)?.label || val
   452→    },
   453→    [],
   454→  )
   455→
   456→  // ==========================================================================
   457→  // Render
   458→  // ==========================================================================
   459→
   460→  return (
   461→    <div className="space-y-6">
   462→      {/* Page Header */}
   463→      <PageHeader
   464→        title="Transactions"
   465→        description="Track and manage all your financial transactions"
   466→        breadcrumbs={[
   467→          { label: 'Dashboard', href: '/dashboard' },
   468→          { label: 'Transactions' },
   469→        ]}
   470→        actions={
   471→          <div className="flex items-center gap-2">
   472→            <DropdownMenu>
   473→              <DropdownMenuTrigger asChild>
   474→                <Button variant="outline" size="sm" disabled={transactions.length === 0}>
   475→                  <Download className="size-4" />
   476→                  <span className="hidden sm:inline">Export</span>
   477→                </Button>
   478→              </DropdownMenuTrigger>
   479→              <DropdownMenuContent align="end">
   480→                <DropdownMenuItem onClick={() => exportTransactions('csv')}>
   481→                  <FileSpreadsheet className="size-4" />
   482→                  Export as CSV
   483→                </DropdownMenuItem>
   484→                <DropdownMenuSeparator />
   485→                <DropdownMenuItem onClick={() => exportTransactions('json')}>
   486→                  <FileJson className="size-4" />
   487→                  Export as JSON
   488→                </DropdownMenuItem>
   489→              </DropdownMenuContent>
   490→            </DropdownMenu>
   491→            <Button size="sm" onClick={openCreateForm}>
   492→              <Plus className="size-4" />
   493→              <span className="hidden sm:inline">Add Transaction</span>
   494→            </Button>
   495→          </div>
   496→        }
   497→      />
   498→
   499→      {/* Filters Bar */}
   500→      <motion.div
   501→        initial={{ opacity: 0, y: -10 }}
   502→        animate={{ opacity: 1, y: 0 }}
   503→        transition={{ duration: 0.3 }}
   504→        className="rounded-lg border bg-card p-4"
   505→      >
   506→        <div className="flex flex-col gap-3 lg:flex-row lg:items-center lg:gap-2">
   507→          {/* Search */}
   508→          <div className="relative flex-1 lg:max-w-xs">
   509→            <Search className="absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   510→            <Input
   511→              placeholder="Search transactions..."
   512→              value={searchInput}
   513→              onChange={(e) => handleSearchChange(e.target.value)}
   514→              className="pl-8"
   515→            />
   516→          </div>
   517→
   518→          {/* Type Filter */}
   519→          <Select
   520→            value={filters.type || 'all'}
   521→            onValueChange={(val) =>
   522→              updateFilter({ type: val === 'all' ? undefined : (val as TransactionFilters['type']) })
   523→            }
   524→          >
   525→            <SelectTrigger className="w-full lg:w-[130px]">
   526→              <SelectValue placeholder="Type" />
   527→            </SelectTrigger>
   528→            <SelectContent>
   529→              <SelectItem value="all">All Types</SelectItem>
   530→              <SelectItem value="income">Income</SelectItem>
   531→              <SelectItem value="expense">Expense</SelectItem>
   532→            </SelectContent>
   533→          </Select>
   534→
   535→          {/* Category Filter */}
   536→          <Select
   537→            value={filters.categoryId || 'all'}
   538→            onValueChange={(val) =>
   539→              updateFilter({ categoryId: val === 'all' ? undefined : val })
   540→            }
   541→          >
   542→            <SelectTrigger className="w-full lg:w-[160px]">
   543→              <SelectValue placeholder="Category" />
   544→            </SelectTrigger>
   545→            <SelectContent>
   546→              <SelectItem value="all">All Categories</SelectItem>
   547→              {categories.map((cat) => (
   548→                <SelectItem key={cat.id} value={cat.id}>
   549→                  <span className="flex items-center gap-1.5">
   550→                    {cat.icon && <span className="text-sm">{cat.icon}</span>}
   551→                    {cat.name}
   552→                  </span>
   553→                </SelectItem>
   554→              ))}
   555→            </SelectContent>
   556→          </Select>
   557→
   558→          {/* Start Date */}
   559→          <Popover open={startOpen} onOpenChange={setStartOpen}>
   560→            <PopoverTrigger asChild>
   561→              <Button
   562→                variant="outline"
   563→                className="w-full justify-start text-left font-normal lg:w-[160px]"
   564→              >
   565→                <CalendarIcon className="mr-2 size-4" />
   566→                {startDate ? format(startDate, 'MMM d, yyyy') : 'From'}
   567→              </Button>
   568→            </PopoverTrigger>
   569→            <PopoverContent className="w-auto p-0" align="start">
   570→              <Calendar
   571→                mode="single"
   572→                selected={startDate}
   573→                onSelect={(d) => {
   574→                  setStartDate(d)
   575→                  setStartOpen(false)
   576→                }}
   577→                initialFocus
   578→              />
   579→            </PopoverContent>
   580→          </Popover>
   581→
   582→          {/* End Date */}
   583→          <Popover open={endOpen} onOpenChange={setEndOpen}>
   584→            <PopoverTrigger asChild>
   585→              <Button
   586→                variant="outline"
   587→                className="w-full justify-start text-left font-normal lg:w-[160px]"
   588→              >
   589→                <CalendarIcon className="mr-2 size-4" />
   590→                {endDate ? format(endDate, 'MMM d, yyyy') : 'To'}
   591→              </Button>
   592→            </PopoverTrigger>
   593→            <PopoverContent className="w-auto p-0" align="start">
   594→              <Calendar
   595→                mode="single"
   596→                selected={endDate}
   597→                onSelect={(d) => {
   598→                  setEndDate(d)
   599→                  setEndOpen(false)
   600→                }}
   601→                initialFocus
   602→              />
   603→            </PopoverContent>
   604→          </Popover>
   605→
   606→          {/* Sort By */}
   607→          <Select
   608→            value={filters.sortBy || 'date'}
   609→            onValueChange={(val) => updateFilter({ sortBy: val })}
   610→          >
   611→            <SelectTrigger className="w-full lg:w-[130px]">
   612→              <ArrowUpDown className="mr-1 size-4" />
   613→              <SelectValue />
   614→            </SelectTrigger>
   615→            <SelectContent>
   616→              {SORT_OPTIONS.map((opt) => (
   617→                <SelectItem key={opt.value} value={opt.value}>
   618→                  {opt.label}
   619→                </SelectItem>
   620→              ))}
   621→            </SelectContent>
   622→          </Select>
   623→
   624→          {/* Sort Order Toggle */}
   625→          <Button
   626→            variant="outline"
   627→            size="icon"
   628→            onClick={toggleSortOrder}
   629→            title={
   630→              filters.sortOrder === 'asc' ? 'Ascending' : 'Descending'
   631→            }
   632→          >
   633→            <SortIcon className="size-4" />
   634→          </Button>
   635→
   636→          {/* Clear Filters */}
   637→          {hasActiveFilters && (
