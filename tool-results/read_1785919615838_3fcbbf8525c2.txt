    94→function getGreeting(): string {

   101→function getHealthInfo(score: number) {
   102→  return (
   103→    FINANCIAL_HEALTH_RANGES.find(
   104→      (r) => score >= r.min && score <= r.max,
   105→    ) ?? FINANCIAL_HEALTH_RANGES[FINANCIAL_HEALTH_RANGES.length - 1]
   106→  )
   107→}

   109→function getBudgetColor(percent: number) {
   110→  if (percent > 90) return { bar: '#f43f5e', text: 'text-rose-600 dark:text-rose-400' }
   111→  if (percent >= 70) return { bar: '#f59e0b', text: 'text-amber-600 dark:text-amber-400' }
   112→  return { bar: '#10b981', text: 'text-emerald-600 dark:text-emerald-400' }
   113→}

   115→function getBudgetGradient(percent: number) {
   116→  if (percent > 90) return 'from-rose-400 to-rose-500'
   117→  if (percent >= 70) return 'from-amber-400 to-amber-500'
   118→  return 'from-emerald-400 to-emerald-500'
