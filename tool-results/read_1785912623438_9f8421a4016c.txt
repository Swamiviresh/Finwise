   105→function getGreeting(): string {
   106→  const hour = new Date().getHours()
   107→  if (hour < 12) return 'Good morning'
   108→  if (hour < 17) return 'Good afternoon'
   109→  return 'Good evening'

   112→function getHealthInfo(score: number) {
   113→  return (
   114→    FINANCIAL_HEALTH_RANGES.find(
   115→      (r) => score >= r.min && score <= r.max,
   116→    ) ?? FINANCIAL_HEALTH_RANGES[FINANCIAL_HEALTH_RANGES.length - 1]
   117→  )

   120→function getBudgetBarClass(percent: number): string {
   121→  if (percent > 90) return '[&>div]:bg-red-500'
   122→  if (percent >= 70) return '[&>div]:bg-amber-500'
   123→  return '[&>div]:bg-emerald-500'

   126→function daysUntil(dateStr: string | Date): number {

   586→  return (

   608→                    percent >= 70 &&
   609→                      percent <= 90 &&
   610→                      'text-amber-600 dark:text-amber-400',
   611→                    percent < 70 && 'text-emerald-600 dark:text-emerald-400',
   612→                  )}
   613→                >
   614→                  {percent}%
   615→                </span>
   616→              </div>
   617→              <Progress
   618→                value={percent}
   619→                className={cn('h-2', getBudgetBarClass(percent))}
   620→              />
   621→              <div className="flex items-center justify-between text-xs text-muted-foreground">
   622→                <span>
   623→                  {formatCurrency(budget.spent)} of {formatCurrency(budget.amount)}

   627→            </div>
   628→          )
   629→        })}

   635→function UpcomingBillsCard({
   636→  subscriptions,
   637→  isLoading,
   638→}: {
   639→  subscriptions: Array<{
   640→    id: string
   641→    name: string
   642→    amount: number
   643→    nextBillingDate: Date | string
   644→    icon: string | null
   645→    color: string
   646→    billingCycle: string
   647→  }>
   648→  isLoading: boolean
   649→}) {
   650→  if (isLoading) {
   651→    return (
   652→      <Card>
   653→        <CardHeader>
   654→          <CardTitle className="text-base">Upcoming Bills</CardTitle>
   655→        </CardHeader>
   656→        <CardContent>
   657→          <ListSkeleton count={4} />
   658→        </CardContent>
   659→      </Card>
   660→    )
   661→  }
   662→
   663→  const upcoming = subscriptions
   664→    .filter((s) => {
   665→      const days = daysUntil(s.nextBillingDate)
   666→      return days >= 0 && days <= 30
   667→    })
   668→    .sort(
   669→      (a, b) =>
   670→        new Date(a.nextBillingDate).getTime() -
   671→        new Date(b.nextBillingDate).getTime(),
   672→    )
   673→    .slice(0, 5)
   674→
   675→  if (upcoming.length === 0) {
   676→    return (
   677→      <Card>
   678→        <CardHeader>
   679→          <CardTitle className="text-base">Upcoming Bills</CardTitle>
   680→        </CardHeader>
   681→        <CardContent>
   682→          <div className="flex flex-col items-center justify-center py-6 text-center">
   683→            <CalendarDays className="mb-3 size-10 text-muted-foreground/50" />
   684→            <p className="text-sm font-medium text-muted-foreground">
   685→              No upcoming bills
   686→            </p>
   687→            <p className="mt-1 text-xs text-muted-foreground/70">
   688→              Your subscription tracking is clear
   689→            </p>
   690→          </div>
   691→        </CardContent>
   692→      </Card>
   693→    )
   694→  }
   695→
   696→  return (
   697→    <Card>
   698→      <CardHeader>
   699→        <CardTitle className="text-base">Upcoming Bills</CardTitle>
   700→        <CardDescription>Next 30 days</CardDescription>
   701→      </CardHeader>
   702→      <CardContent>
   703→        <div className="space-y-2.5">
   704→          {upcoming.map((sub) => {
   705→            const days = daysUntil(sub.nextBillingDate)
   706→            const isSoon = days <= 3
   707→
   708→            return (
   709→              <div
   710→                key={sub.id}
   711→                className="flex items-center gap-3 rounded-lg p-2.5 transition-colors hover:bg-muted/50"
   712→              >
   713→                <div
   714→                  className="flex size-9 shrink-0 items-center justify-center rounded-lg text-base"
   715→                  style={{ backgroundColor: `${sub.color}15` }}
   716→                >
   717→                  {sub.icon ?? <Receipt className="size-4" />}
   718→                </div>
   719→                <div className="min-w-0 flex-1">
   720→                  <p className="truncate text-sm font-medium">{sub.name}</p>
   721→                  <p className="text-xs text-muted-foreground">
   722→                    {formatDate(sub.nextBillingDate)}
   723→                  </p>
   724→                </div>
   725→                <div className="shrink-0 text-right">
   726→                  <p className="text-sm font-semibold tabular-nums">
   727→                    {formatCurrency(sub.amount)}
   728→                  </p>
   729→                  <Badge
   730→                    variant={isSoon ? 'destructive' : 'secondary'}
   731→                    className="mt-0.5 text-[10px]"
   732→                  >
   733→                    {days === 0
   734→                      ? 'Today'
   735→                      : days === 1
   736→                        ? 'Tomorrow'
   737→                        : `${days}d`}
   738→                  </Badge>
   739→                </div>
   740→              </div>
   741→            )
   742→          })}
   743→        </div>
   744→      </CardContent>
   745→    </Card>
   746→  )
   747→}
