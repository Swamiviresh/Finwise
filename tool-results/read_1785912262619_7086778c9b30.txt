    91→function getGreeting(): string {
    92→  const hour = new Date().getHours()
    93→  if (hour < 12) return 'Good morning'
    94→  if (hour < 17) return 'Good afternoon'
    95→  return 'Good evening'
    96→}

    98→function getHealthInfo(score: number) {
    99→  return (
   100→    FINANCIAL_HEALTH_RANGES.find(
   101→      (r) => score >= r.min && score <= r.max,
   102→    ) ?? FINANCIAL_HEALTH_RANGES[FINANCIAL_HEALTH_RANGES.length - 1]
   103→  )
   104→}

   106→function getBudgetColor(percent: number): string {
   107→  if (percent > 90) return 'bg-red-500'
   108→  if (percent >= 70) return 'bg-amber-500'
   109→  return 'bg-emerald-500'
   110→}
   111→
   112→function getBudgetBarClass(percent: number): string {
   113→  if (percent > 90) return '[&>div]:bg-red-500'
   114→  if (percent >= 70) return '[&>div]:bg-amber-500'
   115→  return '[&>div]:bg-emerald-500'
   116→}

   118→function daysUntil(dateStr: string | Date): number {
   119→  const d = typeof dateStr === 'string' ? parseISO(dateStr) : dateStr
   120→  return differenceInDays(d, new Date())
   121→}

   430→function UpcomingBillsCard({
   431→  subscriptions,
   432→  isLoading,
   433→}: {
   434→  subscriptions: Array<{
   435→    id: string
   436→    name: string
   437→    amount: number
   438→    nextBillingDate: Date | string
   439→    icon: string | null
   440→    color: string
   441→    billingCycle: string
   442→  }>
   443→  isLoading: boolean
   444→}) {
   445→  if (isLoading) {
   446→    return (
   447→      <Card>
   448→        <CardHeader>
   449→          <CardTitle className="text-base">Upcoming Bills</CardTitle>
   450→        </CardHeader>
   451→        <CardContent>
   452→          <ListSkeleton count={4} />

   458→  const upcoming = subscriptions
   459→    .filter((s) => {

   464→      (a, b) =>
   465→        new Date(a.nextBillingDate).getTime() -
   466→        new Date(b.nextBillingDate).getTime(),

   471→    return (

   474→          <CardTitle className="text-base">Upcoming Bills</CardTitle>
   475→        </CardHeader>

   477→          <div className="flex flex-col items-center justify-center py-6 text-center">
   478→            <CalendarDays className="mb-3 size-10 text-muted-foreground/50" />
   479→            <p className="text-sm font-medium text-muted-foreground">
   480→              No upcoming bills
   481→            </p>
   482→            <p className="mt-1 text-xs text-muted-foreground/70">
   483→              Your subscription tracking is clear
   484→            </p>
   485→          </div>
   486→        </CardContent>
   487→      </Card>
   488→    )
   489→  }
   490→
   491→  return (
   492→    <Card>
   493→      <CardHeader>
   494→        <CardTitle className="text-base">Upcoming Bills</CardTitle>
   495→        <CardDescription>Next 30 days</CardDescription>
   496→      </CardHeader>
   497→      <CardContent>

   499→          {upcoming.map((sub) => {
   500→            const days = daysUntil(sub.nextBillingDate)
   501→            const isSoon = days <= 3
   502→
   503→            return (
   504→              <div
   505→                key={sub.id}
   506→                className="flex items-center gap-3 rounded-lg p-2.5 transition-colors hover:bg-muted/50"
   507→              >
   508→                <div
   509→                  className="flex size-9 shrink-0 items-center justify-center rounded-lg text-base"
   510→                  style={{ backgroundColor: `${sub.color}15` }}
   511→                >
   512→                  {sub.icon ?? <Receipt className="size-4" />}
   513→                </div>
   514→                <div className="min-w-0 flex-1">
   515→                  <p className="truncate text-sm font-medium">{sub.name}</p>
   516→                  <p className="text-xs text-muted-foreground">
   517→                    {formatDate(sub.nextBillingDate)}
   518→                  </p>
   519→                </div>
   520→                <div className="text-right shrink-0">
   521→                  <p className="text-sm font-semibold tabular-nums">
   522→                    {formatCurrency(sub.amount)}
   523→                  </p>
   524→                  <Badge
   525→                    variant={isSoon ? 'destructive' : 'secondary'}
   526→                    className="mt-0.5 text-[10px]"
   527→                  >
   528→                    {days === 0
   529→                      ? 'Today'
   530→                      : days === 1

   533→                  </Badge>
   534→                </div>
   535→              </div>
