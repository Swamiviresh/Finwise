   281→export function AiAssistantPage() {
   282→  // --- Chat state ---
   283→  const [chats, setChats] = React.useState<LocalChat[]>([])
   284→  const [activeChatId, setActiveChatId] = React.useState<string | null>(null)
   285→  const [input, setInput] = React.useState('')
   286→  const [isTyping, setIsTyping] = React.useState(false)
   287→  const [sidebarOpen, setSidebarOpen] = React.useState(true)
   288→
   289→  const messagesEndRef = React.useRef<HTMLDivElement>(null)
   290→  const textareaRef = React.useRef<HTMLTextAreaElement>(null)
   291→  const chatContainerRef = React.useRef<HTMLDivElement>(null)
   292→
   293→  const activeChat = chats.find((c) => c.id === activeChatId) ?? null
   294→
   295→  // --- Auto-scroll ---
   296→  React.useEffect(() => {
   297→    if (messagesEndRef.current) {
   298→      messagesEndRef.current.scrollIntoView({ behavior: 'smooth' })
   299→    }
   300→  }, [activeChat?.messages, isTyping])
   301→
   302→  // --- Auto-resize textarea ---
   303→  React.useEffect(() => {
   304→    if (textareaRef.current) {
   305→      textareaRef.current.style.height = 'auto'
   306→      textareaRef.current.style.height = `${Math.min(textareaRef.current.scrollHeight, 200)}px`
   307→    }
   308→  }, [input])
   309→
   310→  // --- Generate unique ID ---
   311→  const genId = React.useCallback(() => crypto.randomUUID(), [])
   312→
   313→  // --- Create new chat ---
   314→  const createChat = React.useCallback(() => {
   315→    const newChat: LocalChat = {
   316→      id: genId(),
   317→      title: 'New Conversation',
   318→      messages: [],
   319→      createdAt: new Date(),
   320→      updatedAt: new Date(),
   321→    }
   322→    setChats((prev) => [newChat, ...prev])
   323→    setActiveChatId(newChat.id)
   324→    setInput('')
   325→  }, [genId])
   326→
   327→  // --- Delete chat ---
   328→  const deleteChat = React.useCallback(
   329→    (chatId: string) => {
   330→      setChats((prev) => prev.filter((c) => c.id !== chatId))
   331→      if (activeChatId === chatId) {
   332→        setActiveChatId(null)
   333→      }
   334→    },
   335→    [activeChatId],
   336→  )
   337→
   338→  // --- Send message ---
   339→  const handleSend = React.useCallback(
   340→    async (messageContent?: string) => {
   341→      const content = (messageContent || input).trim()
   342→      if (!content || isTyping) return
   343→
   344→      let chatId = activeChatId
   345→
   346→      // Create a new chat if none active
   347→      if (!chatId) {
   348→        const newChat: LocalChat = {
   349→          id: genId(),
   350→          title: content.slice(0, 50),
   351→          messages: [],
   352→          createdAt: new Date(),
   353→          updatedAt: new Date(),
   354→        }
   355→        setChats((prev) => [newChat, ...prev])
   356→        chatId = newChat.id
   357→        setActiveChatId(chatId)
   358→      }
   359→
   360→      const userMessage: LocalChatMessage = {
   361→        id: genId(),
   362→        role: 'user',
   363→        content,
   364→        createdAt: new Date(),
   365→      }
   366→
   367→      // Add user message and update title on first message
   368→      setChats((prev) =>
   369→        prev.map((c) =>
   370→          c.id === chatId
   371→            ? {
   372→                ...c,
   373→                title: c.messages.length === 0 ? content.slice(0, 50) : c.title,
   374→                messages: [...c.messages, userMessage],
   375→                updatedAt: new Date(),
   376→              }
   377→            : c,
   378→        ),
   379→      )
   380→
   381→      setInput('')
   382→      setIsTyping(true)
   383→
   384→      try {
   385→        // Try the real API first
   386→        const token =
   387→          typeof window !== 'undefined'
   388→            ? localStorage.getItem('token')
   389→            : null
   390→        const headers: Record<string, string> = {
   391→          'Content-Type': 'application/json',
   392→        }
   393→        if (token) headers.Authorization = `Bearer ${token}`
   394→
   395→        // Build message history for API
   396→        const currentChat = chats.find((c) => c.id === chatId)
   397→        const apiMessages = [
   398→          ...(currentChat?.messages.map((m) => ({
   399→            role: m.role,
   400→            content: m.content,
   401→          })) ?? []),
   402→          { role: 'user', content },
   403→        ]
   404→
   405→        const res = await fetch('/api/ai/chat', {
   406→          method: 'POST',
   407→          headers,
   408→          body: JSON.stringify({ messages: apiMessages }),
   409→        })
   410→
   411→        if (res.ok) {
   412→          const data = await res.json()
   413→          const aiContent = data.content || data.message || data.data?.content || ''
   414→          if (aiContent) {
   415→            const assistantMessage: LocalChatMessage = {
   416→              id: genId(),
   417→              role: 'assistant',
   418→              content: aiContent,
   419→              createdAt: new Date(),
   420→            }
   421→            setChats((prev) =>
   422→              prev.map((c) =>
   423→                c.id === chatId
   424→                  ? { ...c, messages: [...c.messages, assistantMessage], updatedAt: new Date() }
   425→                  : c,
   426→              ),
   427→            )
   428→            setIsTyping(false)
   429→            return
   430→          }
   431→        }
   432→        // Fall through to local response
   433→      } catch {
   434→        // Fall through to local response
   435→      }
   436→
   437→      // Fallback to local financial tips
   438→      await new Promise((resolve) => setTimeout(resolve, 800 + Math.random() * 1200))
   439→      const response = getLocalResponse(content)
   440→      const assistantMessage: LocalChatMessage = {
   441→        id: genId(),
   442→        role: 'assistant',
   443→        content: response,
   444→        createdAt: new Date(),
   445→      }
   446→
   447→      setChats((prev) =>
   448→        prev.map((c) =>
   449→          c.id === chatId
   450→            ? { ...c, messages: [...c.messages, assistantMessage], updatedAt: new Date() }
   451→            : c,
   452→        ),
   453→      )
   454→      setIsTyping(false)
   455→    },
   456→    [input, isTyping, activeChatId, chats, genId],
   457→  )
   458→
   459→  // --- Handle keyboard ---
   460→  const handleKeyDown = React.useCallback(
   461→    (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   462→      if (e.key === 'Enter' && !e.shiftKey) {
   463→        e.preventDefault()
   464→        handleSend()
   465→      }
   466→    },
   467→    [handleSend],
   468→  )
   469→
   470→  // --- Empty state ---
   471→  if (!activeChat && chats.length === 0) {
   472→    return (
   473→      <div className="flex h-full">
   474→        {/* Mobile sidebar toggle */}
   475→        <div className="absolute top-4 left-4 z-10 lg:hidden">
   476→          <Button
   477→            variant="ghost"
   478→            size="icon"
   479→            onClick={() => setSidebarOpen(!sidebarOpen)}
   480→          >
   481→            <PanelLeft className="size-5" />
   482→          </Button>
   483→        </div>
   484→
   485→        {/* Welcome screen */}
   486→        <motion.div
   487→          initial={{ opacity: 0, y: 20 }}
   488→          animate={{ opacity: 1, y: 0 }}
   489→          transition={{ duration: 0.5 }}
   490→          className="flex flex-1 flex-col items-center justify-center gap-8 p-6"
   491→        >
   492→          <div className="flex flex-col items-center gap-3">
   493→            <div className="flex size-20 items-center justify-center rounded-2xl bg-primary/10">
   494→              <Sparkles className="size-10 text-primary" />
   495→            </div>
   496→            <div className="text-center">
   497→              <h1 className="text-3xl font-bold tracking-tight">
   498→                AI Financial Assistant
   499→              </h1>
   500→              <p className="mt-2 max-w-md text-muted-foreground">
   501→                Get personalized financial insights, budget recommendations, and
   502→                smart money advice powered by AI.
   503→              </p>
   504→            </div>
   505→          </div>
   506→
   507→          {/* Quick actions */}
   508→          <div className="w-full max-w-2xl">
   509→            <p className="mb-4 text-center text-sm font-medium text-muted-foreground">
   510→              Try asking about:
   511→            </p>
   512→            <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-3">
   513→              {QUICK_ACTIONS.map((action) => {
   514→                const Icon = action.icon
   515→                return (
   516→                  <motion.button
   517→                    key={action.label}
   518→                    whileHover={{ scale: 1.02 }}
   519→                    whileTap={{ scale: 0.98 }}
   520→                    onClick={() => handleSend(action.prompt)}
   521→                    className="group flex items-start gap-3 rounded-xl border bg-card p-4 text-left transition-colors hover:bg-accent"
   522→                  >
   523→                    <div className="flex size-9 shrink-0 items-center justify-center rounded-lg bg-primary/10">
   524→                      <Icon className="size-4 text-primary" />
   525→                    </div>
   526→                    <span className="text-sm font-medium leading-tight">
   527→                      {action.label}
   528→                    </span>
   529→                  </motion.button>
   530→                )
   531→              })}
   532→            </div>
   533→          </div>
   534→
   535→          {/* Input area at bottom */}
   536→          <div className="w-full max-w-2xl">
   537→            <div className="flex items-end gap-2 rounded-2xl border bg-card p-2 shadow-sm">
   538→              <Textarea
   539→                ref={textareaRef}
   540→                value={input}
   541→                onChange={(e) => setInput(e.target.value)}
   542→                onKeyDown={handleKeyDown}
   543→                placeholder="Ask about your finances..."
   544→                className="min-h-[44px] flex-1 resize-none border-0 bg-transparent p-2 text-sm shadow-none focus-visible:ring-0"
   545→                rows={1}
   546→              />
   547→              <Button
   548→                size="icon"
   549→                className="size-10 shrink-0 rounded-xl"
   550→                onClick={() => handleSend()}
   551→                disabled={!input.trim()}
   552→              >
   553→                <Send className="size-4" />
   554→                <span className="sr-only">Send message</span>
   555→              </Button>
   556→            </div>
   557→          </div>
   558→        </motion.div>
   559→      </div>
   560→    )
   561→  }
   562→
   563→  // --- Chat view ---
   564→  return (
   565→    <div className="flex h-full overflow-hidden">
   566→      {/* Sidebar */}
   567→      <AnimatePresence mode="wait">
   568→        {sidebarOpen && (
   569→          <motion.aside
   570→            initial={{ width: 0, opacity: 0 }}
   571→            animate={{ width: 280, opacity: 1 }}
   572→            exit={{ width: 0, opacity: 0 }}
   573→            transition={{ duration: 0.2, ease: 'easeInOut' }}
   574→            className="flex shrink-0 flex-col border-r bg-muted/30 overflow-hidden"
   575→          >
   576→            {/* Sidebar header */}
   577→            <div className="flex items-center justify-between p-3">
   578→              <Button
   579→                variant="ghost"
   580→                size="sm"
   581→                className="gap-1.5"
   582→                onClick={createChat}
   583→              >
   584→                <Plus className="size-4" />
   585→                New Chat
   586→              </Button>
   587→              <Button
   588→                variant="ghost"
   589→                size="icon"
   590→                className="size-8"
   591→                onClick={() => setSidebarOpen(false)}
   592→              >
   593→                <PanelLeftClose className="size-4" />
   594→              </Button>
   595→            </div>
   596→
   597→            {/* Chat list */}
   598→            <ScrollArea className="flex-1 px-2">
   599→              <div className="space-y-1 pb-4">
   600→                {chats.map((chat) => (
   601→                  <motion.div
   602→                    key={chat.id}
   603→                    layout
   604→                    initial={{ opacity: 0, x: -10 }}
   605→                    animate={{ opacity: 1, x: 0 }}
   606→                    exit={{ opacity: 0, x: -10 }}
   607→                    className="group flex items-center gap-2"
   608→                  >
   609→                    <button
   610→                      onClick={() => setActiveChatId(chat.id)}
   611→                      className={cn(
   612→                        'flex flex-1 items-center gap-2 rounded-lg px-3 py-2 text-left text-sm transition-colors hover:bg-accent',
   613→                        activeChatId === chat.id && 'bg-accent font-medium',
   614→                      )}
   615→                    >
   616→                      <MessageSquare className="size-4 shrink-0 text-muted-foreground" />
   617→                      <span className="truncate">{chat.title}</span>
