   220→function SubscriptionIcon({
   221→  name,
   222→  icon,
   223→  color,
   224→  size = 'md',
   225→}: {
   226→  name: string
   227→  icon?: string | null
   228→  color?: string
   229→  size?: 'sm' | 'md' | 'lg'
   230→}) {
   231→  const preset = findPreset(name)
   232→  const resolvedIcon = icon || preset?.icon || name.charAt(0).toUpperCase()
   233→  const resolvedColor = color || preset?.color || '#6b7280'
   234→
   235→  const sizeClass = {
   236→    sm: 'size-8 text-xs',
   237→    md: 'size-10 text-sm',
   238→    lg: 'size-12 text-base',
   239→  }[size]
   240→
   241→  return (
   242→    <div
   243→      className={`${sizeClass} flex shrink-0 items-center justify-center rounded-xl font-bold text-white shadow-sm`}
   244→      style={{ backgroundColor: resolvedColor }}
   245→    >
   246→      {resolvedIcon}
   247→    </div>
   248→  )
   249→}

   251→// ============================================================================
   252→// Countdown Badge
   253→// ============================================================================
   254→
   255→function CountdownBadge({ date }: { date: Date | string }) {
   256→  const days = differenceInDays(parseISO(date as string), new Date())
   257→
   258→  let variant: 'urgent' | 'soon' | 'normal' = 'normal'
   259→  if (days <= 3) variant = 'urgent'
   260→  else if (days <= 7) variant = 'soon'
   261→
   262→  const classMap = {
   263→    urgent: 'bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-400',
   264→    soon: 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-400',
   265→    normal: 'bg-muted text-muted-foreground',
   266→  }
   267→
   268→  const label =
   269→    days < 0
   270→      ? `${Math.abs(days)}d overdue`
   271→      : days === 0
   272→        ? 'Today'
   273→        : days === 1
   274→          ? 'Tomorrow'
   275→          : `${days}d left`
   276→
   277→  return (
   278→    <span
   279→      className={`inline-flex items-center gap-1 rounded-md px-2 py-0.5 text-xs font-medium ${classMap[variant]}`}
   280→    >
   281→      <Clock className="size-3" />
   282→      {label}
   283→    </span>
   284→  )
   285→}

   287→// ============================================================================
   288→// Monthly Equivalent
   289→// ============================================================================
   290→
   291→function getMonthlyEquivalent(
   292→  amount: number,
   293→  billingCycle: 'weekly' | 'monthly' | 'yearly',
   294→): number {
   295→  if (billingCycle === 'weekly') return amount * 4.33
   296→  if (billingCycle === 'yearly') return amount / 12
   297→  return amount
   298→}
   299→
   300→function getPerCycleLabel(
   301→  billingCycle: 'weekly' | 'monthly' | 'yearly',
   302→): string {
   303→  if (billingCycle === 'weekly') return '/wk'
   304→  if (billingCycle === 'yearly') return '/yr'
   305→  return '/mo'
   306→}
