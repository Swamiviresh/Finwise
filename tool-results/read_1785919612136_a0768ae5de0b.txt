    74→    74→// ---------------------------------------------------------------------------
    75→    75→// Change Password Schema
    76→    76→// ---------------------------------------------------------------------------
    77→    77→
    78→    78→const changePasswordSchema = z
    79→    79→  .object({
    80→    80→    currentPassword: z.string().min(1, 'Current password is required'),
    81→    81→    newPassword: z
    82→    82→      .string()
    83→    83→      .min(8, 'Password must be at least 8 characters')
    84→    84→      .regex(/[A-Z]/, 'Must contain an uppercase letter')
    85→    85→      .regex(/[a-z]/, 'Must contain a lowercase letter')
    86→    86→      .regex(/[0-9]/, 'Must contain a number'),
    87→    87→    confirmPassword: z.string(),
    88→    88→  })
    89→    89→  .refine((data) => data.newPassword === data.confirmPassword, {
    90→    90→    message: 'Passwords do not match',
    91→    91→    path: ['confirmPassword'],
    92→    92→  })
    93→    93→
    94→    94→type ChangePasswordValues = z.infer<typeof changePasswordSchema>
