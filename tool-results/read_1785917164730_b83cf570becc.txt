   132→function getProgressColor(percent: number): string {
   133→  if (percent >= 90) return 'bg-red-500'
   134→  if (percent >= 70) return 'bg-amber-500'
   135→  return 'bg-emerald-500'
   136→}
   137→
   138→function getProgressBgColor(percent: number): string {
   139→  if (percent >= 90) return 'bg-red-500/20'
   140→  if (percent >= 70) return 'bg-amber-500/20'
   141→  return 'bg-emerald-500/20'
   142→}
   143→
   144→function getProgressTextColor(percent: number): string {
   145→  if (percent >= 90) return 'text-red-600 dark:text-red-400'
   146→  if (percent >= 70) return 'text-amber-600 dark:text-amber-400'
   147→  return 'text-emerald-600 dark:text-emerald-400'
   148→}
   149→
   150→function getAlertType(percent: number, threshold: number): 'danger' | 'warning' | null {
   151→  if (percent >= 100) return 'danger'
   152→  if (percent >= threshold) return 'warning'
   153→  return null
   154→}

   498→export function BudgetsPage() {
   499→  const { user } = useRouterStore()
   500→  const isMobile = useMediaQuery('(max-width: 768px)')
   501→  const queryClient = useQueryClient()
   502→  const currency = user?.currency || APP_CONFIG.DEFAULT_CURRENCY
   503→
   504→  // ---- Dialog / Sheet state ----
   505→  const [formOpen, setFormOpen] = React.useState(false)
   506→  const [editingBudget, setEditingBudget] = React.useState<Budget | null>(null)
   507→  const [deleteTarget, setDeleteTarget] = React.useState<Budget | null>(null)
   508→
   509→  // ---- Data ----
   510→  const { data, isLoading, isError } = useBudgets({
   511→    limit: APP_CONFIG.MAX_PAGE_SIZE,
   512→  })
   513→
   514→  const categoriesQuery = useQuery<ApiResponse<Category>>({
   515→    queryKey: ['categories'],
   516→    queryFn: () => apiRequest('/api/categories'),
   517→    staleTime: 5 * 60 * 1000,
   518→  })
   519→  const categories = React.useMemo(() => {
   520→    if (categoriesQuery.data?.data) {
   521→      return categoriesQuery.data.data as unknown as Category[]
   522→    }
   523→    return DEFAULT_CATEGORIES.map((c) => ({
   524→      id: c.name.toLowerCase().replace(/\s+/g, '-'),
   525→      name: c.name,
   526→      icon: c.icon,
   527→      color: c.color,
   528→      type: c.type,
   529→      isDefault: true,
   530→      userId: null,
   531→      createdAt: new Date(),
   532→      updatedAt: new Date(),
   533→    })) as Category[]
   534→  }, [categoriesQuery.data])
   535→
   536→  const budgets = data?.data ?? []
   537→
   538→  // ---- Mutations ----
   539→  const createMutation = useCreateBudget()
   540→  const updateMutation = useUpdateBudget()
   541→  const deleteMutation = useDeleteBudget()
   542→
   543→  // ---- Derived ----
   544→  const totalBudget = React.useMemo(
   545→    () => budgets.reduce((sum, b) => sum + b.amount, 0),
   546→    [budgets],
   547→  )
   548→  const totalSpent = React.useMemo(
   549→    () => budgets.reduce((sum, b) => sum + b.spent, 0),
   550→    [budgets],
   551→  )
   552→  const overallPercent =
   553→    totalBudget > 0 ? Math.min((totalSpent / totalBudget) * 100, 100) : 0
   554→
   555→  // ---- Budget alerts ----
   556→  const budgetAlerts = React.useMemo(() => {
   557→    return budgets.filter(
   558→      (b) => b.spent > 0 && (b.spent / b.amount) * 100 >= b.alertThreshold,
   559→    )
   560→  }, [budgets])
   561→
   562→  // ---- Form ----
   563→  const form = useForm<BudgetFormValues>({
   564→    resolver: zodResolver(budgetFormSchema),
   565→    defaultValues: {
   566→      name: '',
   567→      amount: 0,
   568→      period: 'monthly',
   569→      startDate: format(new Date(), 'yyyy-MM-dd'),
   570→      endDate: '',
   571→      alertThreshold: 80,
   572→      categoryBudgets: [],
   573→    },
   574→  })
   575→
   576→  const isEditing = !!editingBudget
   577→  const isPending = createMutation.isPending || updateMutation.isPending
   578→
   579→  // ---- Handlers ----
   580→  const openCreate = React.useCallback(() => {
   581→    setEditingBudget(null)
   582→    form.reset({
   583→      name: '',
   584→      amount: 0,
   585→      period: 'monthly',
   586→      startDate: format(new Date(), 'yyyy-MM-dd'),
   587→      endDate: '',
   588→      alertThreshold: 80,
   589→      categoryBudgets: [],
   590→    })
   591→    setFormOpen(true)
   592→  }, [form])
   593→
   594→  const openEdit = React.useCallback(
   595→    (budget: Budget) => {
   596→      setEditingBudget(budget)
   597→      form.reset({
   598→        name: budget.name,
   599→        amount: budget.amount,
   600→        period: budget.period,
   601→        startDate: format(new Date(budget.startDate), 'yyyy-MM-dd'),
   602→        endDate: budget.endDate
   603→          ? format(new Date(budget.endDate), 'yyyy-MM-dd')
   604→          : '',
   605→        alertThreshold: budget.alertThreshold,
   606→        categoryBudgets: budget.categoryBudgets?.length
   607→          ? budget.categoryBudgets.map((cb) => ({
   608→              categoryId: cb.categoryId,
   609→              amount: cb.amount,
   610→            }))
   611→          : [],
   612→      })
   613→      setFormOpen(true)
   614→    },
   615→    [form],
   616→  )
   617→
   618→  const onSubmit = React.useCallback(
   619→    (values: BudgetFormValues) => {
   620→      const payload: BudgetCreate = {
   621→        name: values.name,
   622→        amount: values.amount,
   623→        period: values.period,
   624→        startDate: values.startDate,
   625→        endDate: values.endDate || undefined,
   626→        alertThreshold: values.alertThreshold,
   627→        categoryBudgets: values.categoryBudgets?.length
   628→          ? values.categoryBudgets.filter((cb) => cb.categoryId)
   629→          : undefined,
   630→      }
   631→      if (isEditing && editingBudget) {
   632→        updateMutation.mutate(
   633→          { id: editingBudget.id, data: payload },
   634→          { onSuccess: () => setFormOpen(false) },
   635→        )
   636→      } else {
   637→        createMutation.mutate(payload, {
   638→          onSuccess: () => setFormOpen(false),
   639→        })
   640→      }
   641→    },
   642→    [isEditing, editingBudget, createMutation, updateMutation],
   643→  )
   644→
   645→  const handleDelete = React.useCallback(() => {
   646→    if (!deleteTarget) return
   647→    deleteMutation.mutate(deleteTarget.id, {
   648→      onSuccess: () => setDeleteTarget(null),
   649→    })
   650→  }, [deleteTarget, deleteMutation])
   651→
   652→  const buildCategoryMap = React.useCallback(() => {
   653→    const map = new Map<string, Category>()
   654→    for (const c of categories) {
   655→      map.set(c.id, c)
   656→    }
   657→    return map
   658→  }, [categories])
   659→
   660→  const categoryMap = buildCategoryMap()
   661→
   662→  // ---- Render ----
   663→  if (isLoading) {
   664→    return (
   665→      <div className="space-y-6">
   666→        <PageHeader title="Budgets" actions={<div className="h-10 w-36" />} />
   667→        <div className="grid gap-4 sm:grid-cols-3">
   668→          <CardSkeleton />
   669→          <CardSkeleton />
   670→          <CardSkeleton />
   671→        </div>
   672→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
