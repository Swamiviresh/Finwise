    74→// ---------------------------------------------------------------------------
    75→// Change Password Schema
    76→// ---------------------------------------------------------------------------
    77→
    78→const changePasswordSchema = z
    79→  .object({
    80→    currentPassword: z.string().min(1, 'Current password is required'),
    81→    newPassword: z
    82→      .string()
    83→      .min(8, 'Password must be at least 8 characters')
    84→      .regex(/[A-Z]/, 'Must contain an uppercase letter')
    85→      .regex(/[a-z]/, 'Must contain a lowercase letter')
    86→      .regex(/[0-9]/, 'Must contain a number'),
    87→    confirmPassword: z.string(),
    88→  })
    89→  .refine((data) => data.newPassword === data.confirmPassword, {
    90→    message: 'Passwords do not match',
    91→    path: ['confirmPassword'],
    92→  })
    93→
    94→type ChangePasswordValues = z.infer<typeof changePasswordSchema>
