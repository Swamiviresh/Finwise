    38→    38→function FadeIn({
    39→    39→  children,
    40→    40→  className,
    41→    41→  delay = 0,
    42→    42→}: {
    43→    43→  children: React.ReactNode
    44→    44→  className?: string
    45→    45→  delay?: number
    46→    46→}) {
    47→    47→  const ref = React.useRef(null)
    48→    48→  const inView = useInView(ref, { once: true, margin: '-60px' })
    49→    49→  return (
    50→    50→    <motion.div
    51→    51→      ref={ref}
    52→    52→      initial={{ opacity: 0, y: 32 }}
    53→    53→      animate={inView ? { opacity: 1, y: 0 } : { opacity: 0, y: 32 }}
    54→    54→      transition={{ duration: 0.6, delay, ease: [0.21, 0.47, 0.32, 0.98] }}
    55→    55→      className={className}
    56→    56→    >
    57→    57→      {children}
    58→    58→    </motion.div>
    59→    59→  )
    60→    60→}
